# AniTalker pip deps. Torch 2.0.1+cu118 comes from env.yml; this file
# handles everything else. Mirrors upstream's README pins with a handful of
# small adjustments documented inline.

# ---------------------------------------------------------------------------
# AniTalker-specific pins (from upstream README)
# ---------------------------------------------------------------------------
# `transformers==4.19.2` is upstream's pin — the HuBERT feature extractor
# in demo.py imports APIs that shifted in later versions. Required.
transformers==4.19.2

# huggingface_hub — 0.10.x is the newest release that still ships the
# `cached_download` symbol transformers 4.19 expects; newer releases removed
# it and break import. `[cli]` extra for the `huggingface-cli` command
# setup_env.sh uses to fetch ckpts from the `taocode/anitalker_ckpts` repo.
huggingface_hub[cli]==0.10.1

# ---------------------------------------------------------------------------
# Audio / video / image I/O
# ---------------------------------------------------------------------------
librosa==0.9.2
soundfile
moviepy==1.0.3          # demo.py uses moviepy for the SR mp4 mux path
imageio==2.19.3
imageio-ffmpeg==0.4.7
opencv-python==4.7.0.72
scikit-image
# python_speech_features is imported unconditionally at the top of demo.py;
# the MFCC code path is gated on --infer_type=mfcc_*, but the import itself
# fires on every run, so the package must be present even when we use the
# hubert_audio_only path.
python_speech_features

# ---------------------------------------------------------------------------
# Speech model toolkit (Conformer encoder used by model/seq2seq.py)
# ---------------------------------------------------------------------------
# espnet 202301 matches upstream's pin. Provides
# `espnet.nets.pytorch_backend.conformer.encoder.Encoder` which AniTalker's
# DiffusionPredictor builds on. Newer espnet releases reorganized this
# import path (espnet2.*) and break the upstream code unchanged.
espnet==202301

# ---------------------------------------------------------------------------
# Numeric stack
# ---------------------------------------------------------------------------
numpy==1.23.5
scipy>=1.10.1

# ---------------------------------------------------------------------------
# Face-SR (optional; enabled by --face_sr to upscale AniTalker's native
# 256×256 output to 512×512 so it matches every other baseline's surface).
# GFPGAN pulls in basicsr + facexlib + realesrgan as transitive deps.
# ---------------------------------------------------------------------------
gfpgan==1.3.8
basicsr==1.4.2

# ---------------------------------------------------------------------------
# Misc utilities AniTalker's code imports
# ---------------------------------------------------------------------------
einops
tqdm
omegaconf
# 1.5.10 has malformed PEP 440 metadata (`torch (>=1.7.*)`) that pip>=24.1
# rejects. The .post0 re-release fixes the metadata; code is identical.
pytorch-lightning==1.5.10.post0
pyyaml
